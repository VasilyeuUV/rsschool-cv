JavaScript · Snail Sort

Codewars kata: convert an N × N array into a single array by traversing its elements clockwise in a spiral order.

const snail = function (array) {
  const result = [];

  while (array.length) {
    result.push(array.shift());

    if (!array.length)
      break;

    array = array.map((_, index) =>
      array.map(row => row[row.length - index - 1])
    );
  }

  return [].concat(...result);
};
